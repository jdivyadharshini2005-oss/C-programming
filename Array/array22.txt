#include<stdio.h>
int main() {
    int size;
    scanf("%d",&size);
    int arr[size];
    for(int i=0;i<size;i++) {
        scanf("%d",&arr[i]);
    }
    int index=0;
    int count=0;
    for(int i=index;i<size-2;i++) {
        if (arr[i] > arr[i-1]){
            if(arr[i] > arr[i+1]) {
                count++;
            }
        }
    }
    printf("%d",count);
}
