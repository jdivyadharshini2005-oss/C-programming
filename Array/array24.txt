#include <stdio.h>
int main() {
    int size;
    scanf("%d", &size);
    int arr[size];
    for (int i = 0; i < size; i++) {
        scanf("%d", &arr[i]);
    }
    int unique = 1;
    for (int i = 0; i < size; i++) {
        int count = 0;  
        for (int j = 0; j < size; j++) {
            if (arr[i] == arr[j]) {
                count++;
            }
        }
        if (count>1) {
            unique = 0;
            break;
        }
    }
    if (unique)
        printf("yes");
    else
        printf("no");

    return 0;
}
