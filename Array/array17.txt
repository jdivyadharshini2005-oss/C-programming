#include<stdio.h>
int main() {
    int size;
    scanf("%d",&size);
    int arr[size];
    for (int i=0;i<size;i++) {
        scanf("%d",&arr[i]);
    }

    int palindrome=1;
    for (int i=0;i<size;i++) {
        int temp=arr[i];
        int rev=0;
        while (temp>0){
            int digit=temp%10;
            rev=rev*10+digit;
            temp=temp/10;
        }
        if (rev!=arr[i]) {
            palindrome=0;
            break;
        }
    }
    if (palindrome) {
        printf("yes");
    }
    else {
        printf("no");
    }
}
