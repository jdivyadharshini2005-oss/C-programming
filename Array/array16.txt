 #include <stdio.h>
int main() {
    int size;
    scanf("%d", &size);
    int arr[size];
    for(int i = 0; i < size; i++) {
        scanf("%d", &arr[i]);
    }
    int minprice = arr[0];
    int maxprofit = arr[1]-arr[0];
    for(int i=0;i<size;i++){
        int profit=arr[i]-minprice;
        if(profit>maxprofit){
            maxprofit=profit;
        }
        if(arr[i]<minprice){
            minprice=arr[i];
        }
    }
    printf("%d",maxprice);
    return 0;
}
