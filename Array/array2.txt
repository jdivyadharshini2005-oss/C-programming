#include<stdio.h>
int main(){
    int size,x,count=0;
    scanf("%d",&size);
    int arr[size];
    for(int i=0;i<size;i++){
        scanf("%d",&arr[i]);
    }
    scanf("%d",&x);
    for(int i=0;i<size;i++){
        if(arr[i]==x){
            count++;
        }
    }
    printf("%d",count);
    return 0;
    
}
