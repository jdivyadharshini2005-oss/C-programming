#include <stdio.h>
#include <limits.h>   
int main() {
    int size;
    scanf("%d", &size);
    int arr[size];
    for(int i = 0; i < size; i++) {
        scanf("%d", &arr[i]);
    }
    int min = INT_MAX;   
    for(int i = 0; i < size; i++) {
        if(arr[i] > 0 && arr[i] < min) {
       min = arr[i];
        }
    }

    if(min == INT_MAX) {
        printf("No positive");
    } 
    else {
        printf("%d", min);
    }

    return 0;
}
