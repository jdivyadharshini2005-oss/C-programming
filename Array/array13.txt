#include<stdio.h>
int main(){
    int size;
    scanf("%d",&size);
    int arr[size];
    for(int i=0;i<size;i++){
        scanf("%d",&arr[i]);
    }
    int sum=0;
    for(int i=0;i<size;i++){
        scanf("%d",&arr[i]);
        sum=sum+arr[i];
    }
    int count=0;
    int average=sum/size;
    for(int i=0;i<size;i++){
        if(arr[i]>average){
            count++;
    }
    }
    printf("%d",count);
    return 0;
}
