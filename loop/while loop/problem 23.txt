#include <stdio.h>
int main() {
    int fuel, N;
    scanf("%d", &fuel);
    scanf("%d", &N);
    int burn[N];
    for (int i = 0; i < N; i++) {
        scanf("%d", &burn[i]);
    }
    int i = 0, emergency = -1;
    while (i < N) {
        fuel -= burn[i];
        if (fuel < 0) {
            emergency = i + 1;
            break;
        }
        i++;
    }
    if (emergency == -1) {
        printf("Emergency Stage: Not Occurred\n");
        printf("Fuel Wasted: 0\n");
    } else {
        printf("Emergency Stage: %d\n", emergency);
        printf("Fuel Wasted: %d\n", -fuel);
    }

    return 0;
}
