#include <stdio.h>
int main() {
    int N;
    scanf("%d", &N);
    int amt[N];
    for (int i = 0; i < N; i++) {
        scanf("%d", &amt[i]);
    }
    int i = 0, highCount = 0, consec = 0;
    int fraudAt = -1;
    while (i < N) {
        if (amt[i] >= 50000) {
            highCount++;
            consec++;
            if (consec == 3 && fraudAt == -1) {
                fraudAt = i + 1;
            }
        } else {
            consec = 0;
        }
        i++;
    }
    if (fraudAt == -1)
        printf("Fraud Triggered At Attempt: Not Triggered\n");
    else
        printf("Fraud Triggered At Attempt: %d\n", fraudAt);

    printf("High-Value Transactions: %d\n", highCount);
    return 0;
}
