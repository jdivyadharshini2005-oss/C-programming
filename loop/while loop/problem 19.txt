#include <stdio.h>
int main() {
    int N;
    scanf("%d", &N);
    int arr[N];
    for (int i = 0; i < N; i++) {
        scanf("%d", &arr[i]);
    }
    int i = 1, drops = 0, consec = 0;
    int crashDay = -1;
    while (i < N) {
        if (arr[i] < arr[i - 1]) {
            drops++;
            consec++;
            if (consec == 3 && crashDay == -1) {
                crashDay = i + 1;
            }
        } else {
            consec = 0;
        }
        i++;
    }
    if (crashDay == -1)
        printf("Crash Day: Not Detected\n");
    else
        printf("Crash Day: %d\n", crashDay);

    printf("Total Drops: %d\n", drops);
    return 0;
}
