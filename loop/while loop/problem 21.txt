#include <stdio.h>
int main() {
    int N;
    scanf("%d", &N);

    int attempts[N];
    for (int i = 0; i < N; i++) {
        scanf("%d", &attempts[i]);
    }

    int i = 0, failed = 0, consecFail = 0;
    int lockAt = -1;

    while (i < N) {
        if (attempts[i] == 0) {
            failed++;
            consecFail++;
            if (consecFail == 3 && lockAt == -1) {
                lockAt = i + 1;
            }
        } else {
            consecFail = 0;
        }
        i++;
    }
    if (lockAt == -1)
        printf("Lock Triggered At Attempt: Not Locked\n");
    else
        printf("Lock Triggered At Attempt: %d\n", lockAt);

    printf("Total Failed Attempts: %d\n", failed);
    return 0;
}
