#include<stdio.h>
int main() {
    int maxcapacity;
    scanf("%d",&maxcapacity);
    int n;
    scanf("%d",&n);
    int patients;
    int i=0;
    int sum=0;
    while (i<n) {
        scanf("%d",&patients);
        sum=sum+patients;
        if (maxcapacity<sum) {
            sum=sum-maxcapacity;
        }
        i++;
    }
    printf("%d\n",maxcapacity);
    printf("%d",sum);
    return 0;
}
